# The address of the lobby service is not part of the source: put it in local/directory.txt (git-ignored) and it is built into the game, so
# players never type it. Same for local/rendezvous.txt (an optional relay server). Without the files the game starts with no lobby service.
foreach(kind directory rendezvous)
  set(file "${CMAKE_SOURCE_DIR}/local/${kind}.txt")
  string(TOUPPER ${kind} KIND)
  if(EXISTS "${file}")
    file(STRINGS "${file}" value LIMIT_COUNT 1)
    string(STRIP "${value}" value)
    if(value)
      list(APPEND CB2_DEFAULT_DEFS "CB2_DEFAULT_${KIND}=\"${value}\"")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${file}")
  endif()
endforeach()
